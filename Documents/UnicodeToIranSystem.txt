        public string Convert(string unicodeStr, bool IsDate, bool RTL)
        {
            var bytes = ConvertToByte(unicodeStr, IsDate, RTL);
            return Encoding.Default.GetString(bytes);
        }

        public byte[] ConvertByte(string unicodeStr, bool IsDate, bool RTL)
        {
            return ConvertToByte(unicodeStr, IsDate, RTL);
        }

        public byte[] ConvertLineWithComma(string unicodeStrWithComma, bool IsDate, bool RTL)
        {
            _isDate = IsDate;
            _rtl = RTL;
            var b = ","u8.ToArray();
            var separator = new char[] { ',' };
            var strArray = unicodeStrWithComma.Split(separator);
            var a = Array.Empty<byte>();
            foreach (var str in strArray)
            {
                if (a.Length != 0)
                {
                    a = ConcateByteArray(a, b);
                }
                a = ConcateByteArray(a, new UnicodeToIranSystem().ConvertByte(str, false, true));
            }
            return a;
        }

        private List<byte> ConvertReverse(byte[] b)
        {
            var list = new List<byte>();
            for (var i = b.Length - 1; i >= 0; i--)
            {
                list.Add(b[i]);
            }
            return list;
        }

        public void SaveAsIranSystem(string[] strArr, string path)
        {
            try
            {
                var num = 1;
                using var writer = new BinaryWriter(new FileStream(path, FileMode.Create));
                foreach (var str in strArr)
                {
                    var buffer = ConvertToByte(str, false, true);
                    writer.Write(buffer);
                    if (num != strArr.Length)
                    {
                        writer.Write(System.Convert.ToByte(13));
                        writer.Write(System.Convert.ToByte(10));
                    }
                    num++;
                }
                writer.Close();
            }
            catch (Exception)
            {
                throw;
            }
        }

        private byte[] ConcateByteArray(byte[] a, byte[] b)
        {
            int num2;
            var buffer = new byte[a.Length + b.Length];
            var index = 0;
            for (num2 = 0; num2 < a.Length; num2++)
            {
                buffer[index] = a[num2];
                index++;
            }
            var length = a.Length;
            for (num2 = 0; num2 < b.Length; num2++)
            {
                buffer[length] = b[num2];
                length++;
            }
            return buffer;
        }

        private byte[] Convert(byte[] ca, bool isDate)
        {
            var chArray = new char[2];
            var num = 0;
            var flag = true;
            var list = new List<byte>();
            var list2 = new List<byte>();
            if (_rtl)
            {
                ConvertRtl(ca, chArray, ref num, ref flag, list, list2);
            }
            else if (isDate)
            {
                ConvertDate(ca, list);
            }
            else
            {
                for (var num2 = 0; num2 < ca.Length; num2++)
                {
                    list.Add(ca[num2]);
                }
            }
            return list.ToArray();
        }

        private int ConvertRtl(byte[] ca, char[] chArray, ref int num, ref bool flag, List<byte> list, List<byte> list2)
        {
            int num2;
            for (num2 = ca.Length - 1; num2 >= 0; num2--)
            {
                if (((((ca[num2].ToString().CompareTo("47") >= 0) && (ca[num2].ToString().CompareTo("57") <= 0)) ||
                      ((ca[num2].ToString().CompareTo("65") >= 0) && (ca[num2].ToString().CompareTo("90") <= 0))) ||
                     (((ca[num2].ToString().CompareTo("97") >= 0) && (ca[num2].ToString().CompareTo("122") <= 0)) ||
                      ((ca[num2].ToString().CompareTo("128") >= 0) && (ca[num2].ToString().CompareTo("137") <= 0)))) ||
                    ((list2.Count > 0) && (ca[num2].ToString() == "44")))
                {
                    list2.Add(ca[num2]);
                    flag = false;
                }
                else if (ca[num2].ToString() == "10")
                {
                    if (!flag)
                    {
                        list.AddRange(ConvertReverse(list2.ToArray()));
                        list2.Clear();
                        flag = true;
                    }
                    list.Add(System.Convert.ToByte(10));
                    num2--;
                }
                else
                {
                    if (!flag)
                    {
                        list.AddRange(ConvertReverse(list2.ToArray()));
                        list2.Clear();
                        flag = true;
                    }
                    list.Add(ca[num2]);
                    num++;
                    if (chArray[1] == ' ')
                    {
                        list.Add(System.Convert.ToByte(_ht[System.Convert.ToString(chArray[1])]));
                        num++;
                    }
                }
            }
            list.AddRange(ConvertReverse(list2.ToArray()));
            return num2;
        }

        private static int ConvertDate(byte[] ca, List<byte> list)
        {
            int num2;
            var stack = new Stack<int>();
            for (num2 = 0; num2 < ca.Length; num2++)
            {
                if (ca[num2] != 0x2f)
                {
                    stack.Push(ca[num2]);
                }
                else if (stack.Count > 0)
                {
                    while (stack.Count > 0)
                    {
                        list.Add(System.Convert.ToByte(stack.Pop()));
                    }
                    list.Add(System.Convert.ToByte(0x2f));
                }
                if ((stack.Count > 0) && (num2 == (ca.Length - 1)))
                {
                    while (stack.Count > 0)
                    {
                        list.Add(System.Convert.ToByte(stack.Pop()));
                    }
                }
            }

            return num2;
        }

        private byte[] ConvertToByte(string unicodeStr, bool IsDate, bool RTL)
        {
            _isDate = IsDate;
            _rtl = RTL;
            if (!string.IsNullOrEmpty(unicodeStr))
            {
                unicodeStr = unicodeStr.Replace(((char)(8207)).ToString(), "").Replace(((char)(8206)).ToString(), "");
            }

            var list = new List<byte>();
            for (var i = 0; i < unicodeStr.Length; i++)
            {
                var ch = unicodeStr[i];
                var before = i == 0 ? ' ' : unicodeStr[i - 1];
                var after = i == unicodeStr.Length - 1 ? ' ' : unicodeStr[i + 1];
                var key = FindKey(ch.ToString(), before, after);
                if (!char.Equals(ch, ' ') && (before.ToString() + ch) != "لا")
                {
                    list.Add(System.Convert.ToByte(key));
                }
                else if (char.Equals(ch, ' '))
                {
                    list.Add(System.Convert.ToByte(' '));
                }
            }
            return Convert(list.ToArray(), _isDate);
        }

        private bool IsEnglishCharOrNumber(char? c)
        {
            if (!c.HasValue) return false;

            if (('a' <= c && c <= 'z') ||
                ('A' <= c && c <= 'Z'))
                return true;

            return char.IsDigit(c.Value);
        }

        private object FindKey(string item, char? before = null, char? after = null)
        {
            var key = item;
            if (before.HasValue && (before == ' ' || _seperatedBefore.Contains(before.ToString())
                                    || IsEnglishCharOrNumber(before)))
                key = " " + key;
            if (after.HasValue && (after == ' ' || IsEnglishCharOrNumber(after)))
                key += " ";

            if (item + after == "لا")
            {
                key = item + after;
            }

            var result = _ht[key];
            return result ?? _ht[item];
        }
